\begin{figure}[t]
  \centering
  \includegraphics[width=\textwidth]{figures/framework}
  \caption{One step of the loop. The agent observes the current point cloud
  through a fixed-size state $s_k$, selects a walker and a displacement, and
  proposes a new space--time point. The PDE-constrained linear system of
  \cref{sec:recon} reconstructs the solution there from its neighbours; if
  the stencil is admissible the point joins the cloud and the state advances
  to $s_{k+1}$, otherwise the walker does not move. The weights $\bw$ are
  never learned --- they are fixed by the moment conditions --- so the
  agent's only influence is on \emph{where} points are placed.}
  \label{fig:framework}
\end{figure}
